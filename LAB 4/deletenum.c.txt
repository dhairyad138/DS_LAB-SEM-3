//WAP to delete a number from given location in array

#include <stdio.h>
int main()
{
    int arr[100], n, loc, i;
    printf("Enter the number of elements in the array: ");
    scanf("%d", &n);        
    printf("Enter the elements in the array: ");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }
    printf("Enter the location from which the number is to be deleted: ");
    scanf("%d", &loc);
    if (loc < 0 || loc >= n)
    {
        printf("Invalid location. Please enter a valid location.\n");
        return 1; 
    }
    for (i = loc; i < n - 1; i++)
    {
        arr[i] = arr[i + 1];
    }
    n--; 
    printf("The array after deletion is: ");
    for (i = 0; i < n; i++)
    {
        printf("%d ", arr[i]);
    }
    printf("\n");
    return 0;
}
