//WAP to delete number from sorted array.

#include <stdio.h>
int main()
    {
        int arr[100], n, num, i, j;
        printf("Enter the number of elements in the array: ");
        scanf("%d", &n);
        printf("Enter the elements in sorted order: ");
        for (i = 0; i < n; i++) 
        {
            scanf("%d", &arr[i]);
        }
        printf("Enter the number to be deleted: ");
        scanf("%d", &num);
        for (i = 0; i < n; i++)
        {
            if (arr[i] == num)
            {
                for (j = i; j < n - 1; j++)
                {   
                    arr[j] = arr[j + 1];
                }
                n--;
                break;

    }
